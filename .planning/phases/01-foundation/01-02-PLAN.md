   - sequence_number: INTEGER (reference from pool webhook)
   - share_difficulty: REAL NOT NULL (difficulty value)
   - billable: INTEGER NOT NULL (1 if billable, 0 if not)
   - shares_consumed: INTEGER NOT NULL (1-5 based on priority)
   - FOREIGN KEY (user_id) REFERENCES users(user_id)

Store schema as constants: USERS_TABLE_SQL, TRANSACTIONS_TABLE_SQL, SHARE_EVENTS_TABLE_SQL.
