```python
import httpx
from typing import Optional
import logging

logger = logging.getLogger(__name__)

class PoolClient:
    """Async HTTP client for SV2 pool API integration.

    Handles coinbase address updates during user rotation.
    """

    def __init__(self, pool_url: str, timeout: float = 5.0):
        """Initialize pool client.

        Args:
            pool_url: Base URL of the pool (e.g., http://localhost:8080)
            timeout: Request timeout in seconds (default 5s)
        """
        self.pool_url = pool_url.rstrip('/')
        self.timeout = timeout
        self._client: Optional[httpx.AsyncClient] = None

    async def __aenter__(self):
        """Async context manager entry."""
        self._client = httpx.AsyncClient(timeout=self.timeout)
        return self

    async def __aexit__(self, exc_type, exc_val, exc_tb):
        """Async context manager exit."""
        if self._client:
            await self._client.aclose()

    async def update_coinbase(
        self,
        address: str,
        user_id: int,
        tag: Optional[str] = None
    ) -> bool:
        """Update pool's coinbase address for next mining block.

        Args:
            address: Bitcoin address to mine to
            user_id: User ID for tracking
            tag: Optional user tag for pool logs

        Returns:
            True if update successful, False if failed

        Raises:
            RuntimeError: If client not initialized via context manager
        """
        if not self._client:
            raise RuntimeError("PoolClient must be used as async context manager")

        endpoint = f"{self.pool_url}/api/coinbase"
        payload = {
            "address": address,
            "user_id": user_id,
            "tag": tag
        }

        try:
            response = await self._client.post(endpoint, json=payload)
            response.raise_for_status()
            logger.info(f"Updated coinbase address to {address} (user {user_id})")
            return True

        except httpx.HTTPError as e:
            logger.error(f"Failed to update coinbase address: {e}")
            return False
```

Key design choices:
- Async context manager pattern for clean lifecycle
- Graceful error handling: log and return False, don't crash
- Configurable timeout (5s default)
- Type hints throughout
- Logger for observability
