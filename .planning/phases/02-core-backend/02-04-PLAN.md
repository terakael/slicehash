**1. RotationState dataclass:**

```python
from dataclasses import dataclass
from datetime import datetime
from typing import Optional

@dataclass
class RotationState:
    """Current rotation state (in-memory tracking).

    Attributes:
        current_user_id: User currently mining (None if no users active)
        shares_this_turn: How many shares current user has found this turn
        rotation_started_at: When current user's turn began
        last_rotation_at: When we last rotated (for interval calculation)
    """
    current_user_id: Optional[int] = None
    shares_this_turn: int = 0
    rotation_started_at: Optional[datetime] = None
    last_rotation_at: Optional[datetime] = None
```

**2. calculate_rotation_interval(active_user_count: int) -> float:**
- Returns 60.0 / max(active_user_count, 1) in seconds
- Adaptive: more users = shorter intervals
- Minimum interval 1s (when 60+ users)
- Example: 5 users = 12s, 20 users = 3s, 60 users = 1s

**3. should_rotate(state: RotationState, rotation_interval: float, now: datetime) -> bool:**
- Returns False if current_user_id is None (no rotation needed)
- Returns False if shares_this_turn == 0 (need at least 1 share)
- Returns False if rotation_started_at is None (safety check)
- elapsed = (now - rotation_started_at).total_seconds()
- Returns True if elapsed >= rotation_interval AND shares_this_turn >= 1
- Implements ROT-03: rotation triggers on 1+ share AND time elapsed

**4. select_next_user(db: aiosqlite.Connection, exclude_user_id: Optional[int] = None) -> Optional[int]:**
- Fetch active users via get_active_users(db)
- Filter out exclude_user_id if provided
- If no users available, return None
- Query last_served_at for each active user from database
- Calculate weighted_wait_time = time_since_last_served / priority_multiplier
  - Longer wait = higher priority
  - Higher multiplier = lower priority (they pay more shares)
- Return user_id with maximum weighted_wait_time
- If user never served before (last_served_at NULL), treat as infinite wait (highest priority)
- Implements ROT-01: least recently served with priority weighting

Key SQL query pattern:
```sql
SELECT user_id, last_served_at, priority_multiplier
FROM users
WHERE user_id IN (active_user_ids)
ORDER BY
  CASE
    WHEN last_served_at IS NULL THEN 1  -- Never served = highest priority
    ELSE 0
  END DESC,
  (strftime('%s', 'now') - strftime('%s', last_served_at)) / priority_multiplier DESC
LIMIT 1
```

```python
import asyncio
from datetime import datetime, timedelta
from slicehash.rotation import RotationState, select_next_user, should_rotate, calculate_rotation_interval
from slicehash.db.manager import DatabaseManager, init_database, get_or_create_user, add_transaction

async def test():
    # Initialize test database
    await init_database("test_rotation.db")
    async with DatabaseManager("test_rotation.db") as db:
        # Create 3 users with transactions
        user1 = await get_or_create_user(db, "bc1user1", "user1")
        user2 = await get_or_create_user(db, "bc1user2", "user2")
        user3 = await get_or_create_user(db, "bc1user3", "user3")

        await add_transaction(db, user1, 1000)
        await add_transaction(db, user2, 1000)
        await add_transaction(db, user3, 1000)

        # Set user2 priority to 5 (lower fairness priority)
        await db.execute("UPDATE users SET priority_multiplier = 5 WHERE user_id = ?", (user2,))
        await db.commit()

        # Mark user1 as served recently
        await db.execute(
            "UPDATE users SET last_served_at = datetime('now', '-5 minutes') WHERE user_id = ?",
            (user1,)
        )
        await db.commit()

        # Test: Should select user3 (never served) over user2 (high priority) over user1 (recent)
        next_user = await select_next_user(db)
        print(f"Selected user: {next_user}")
        assert next_user == user3, "Should select never-served user first"

        # Mark user3 as just served
        await db.execute(
            "UPDATE users SET last_served_at = datetime('now') WHERE user_id = ?",
            (user3,)
        )
        await db.commit()

        # Test: Should now select user2 despite high priority (longer wait than user1)
        next_user = await select_next_user(db)
        print(f"Selected user after user3 served: {next_user}")

    # Test rotation interval calculation
    assert calculate_rotation_interval(5) == 12.0
    assert calculate_rotation_interval(20) == 3.0
    assert calculate_rotation_interval(60) == 1.0
    print("Rotation interval tests passed")

    # Test should_rotate logic
    now = datetime.now()
    state = RotationState(
        current_user_id=1,
        shares_this_turn=1,
        rotation_started_at=now - timedelta(seconds=10),
        last_rotation_at=now - timedelta(seconds=15)
    )

    assert should_rotate(state, 5.0, now) is True  # 10s > 5s interval
    assert should_rotate(state, 15.0, now) is False  # 10s < 15s interval

    state_no_shares = RotationState(
        current_user_id=1,
        shares_this_turn=0,
        rotation_started_at=now - timedelta(seconds=10)
    )
    assert should_rotate(state_no_shares, 5.0, now) is False  # No shares yet

    print("All rotation tests passed!")

asyncio.run(test())
```
